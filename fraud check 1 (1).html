# Strategy A: Baseline (no imbalance handling)
pipe_base = Pipeline([
    ('prep', preprocessor),
    ('clf', RandomForestClassifier(n_estimators=200,
                                   random_state=RANDOM_STATE))
])

# Strategy B: class_weight='balanced'
pipe_bal = Pipeline([
    ('prep', preprocessor),
    ('clf', RandomForestClassifier(n_estimators=200,
                                   random_state=RANDOM_STATE,
                                   class_weight='balanced'))
])

# Strategy C: SMOTE
pipe_smote = ImbPipeline([
    ('prep', preprocessor),
    ('smote', SMOTE(random_state=RANDOM_STATE)),
    ('clf', RandomForestClassifier(n_estimators=200,
                                   random_state=RANDOM_STATE))
])

models = {
    'Baseline RF'               : pipe_base,
    'RF + class_weight=balanced': pipe_bal,
    'RF + SMOTE'                : pipe_smote
}

rows = []
for name, pipe in models.items():
    pipe.fit(X_train, y_train)
    pred = pipe.predict(X_test)
    rows.append({
        'Model'           : name,
        'Accuracy'        : round(accuracy_score(y_test, pred), 4),
        'Precision-Risky' : round(precision_score(y_test, pred, pos_label='Risky'), 4),
        'Recall-Risky'    : round(recall_score(y_test, pred, pos_label='Risky'), 4),
        'F1-Risky'        : round(f1_score(y_test, pred, pos_label='Risky'), 4),
    })

comparison = pd.DataFrame(rows).set_index('Model')
print(comparison)

from sklearn.metrics import make_scorer, f1_score

# ---- Impurity-based importance (encoded features) ----
ohe = best_rf.named_steps['prep'].named_transformers_['nom']
ohe_names = ohe.get_feature_names_out(nominal_cols).tolist()
encoded_feature_names = numeric_cols + ohe_names      # length = 6

imp_impurity = pd.Series(
    best_rf.named_steps['clf'].feature_importances_,
    index=encoded_feature_names
).sort_values(ascending=False)

# ---- Permutation importance (RAW features, since best_rf is a pipeline) ----
f1_risky = make_scorer(f1_score, pos_label='Risky')

perm = permutation_importance(
    best_rf, X_test, y_test,
    n_repeats=20,
    random_state=RANDOM_STATE,
    scoring=f1_risky
)

raw_feature_names = X_test.columns.tolist()           # length = 5
imp_perm = pd.Series(perm.importances_mean, index=raw_feature_names)\
             .sort_values(ascending=False)

# ---- Plot side by side ----
fig, axes = plt.subplots(1, 2, figsize=(15,5))

imp_impurity.plot(kind='barh', ax=axes[0], color='crimson')
axes[0].set_title('Impurity Importance (encoded features)')
axes[0].invert_yaxis()

imp_perm.plot(kind='barh', ax=axes[1], color='purple')
axes[1].set_title('Permutation Importance (raw features, F1-Risky)')
axes[1].invert_yaxis()

plt.tight_layout()
plt.show()

print("\nEncoded feature importances (impurity):")
print(imp_impurity.round(4))
print("\nRaw feature importances (permutation):")
print(imp_perm.round(4))
